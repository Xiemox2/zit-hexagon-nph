# NPG bench native library (MIT). Reuses ggml + ggml-hexagon from the sd.cpp fork used by ZIT NPH (MIT).
cmake_minimum_required(VERSION 3.22)
project(npg_native C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(SDCPP_DIR "/workspace/zit-qnn/sdbench/sdcpp" CACHE PATH "sd.cpp fork (read-only use)")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(ANDROID)
  set(GGML_HEXAGON ON CACHE BOOL "" FORCE)
  set(PREBUILT_LIB_DIR "android_aarch64" CACHE STRING "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDCPP_DIR}/ggml" ggml EXCLUDE_FROM_ALL)

set(NPG_SRC npg_bench.cpp npg_common.cpp npg_kernels_src.cpp)
set(CL_INC "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/OpenCL-Headers-main")

if(ANDROID)
  add_library(npgbench SHARED ${NPG_SRC})
  target_compile_definitions(npgbench PRIVATE NPG_HEXAGON=1)
  target_include_directories(npgbench PRIVATE ${CL_INC} "${SDCPP_DIR}/ggml/include")
  target_link_libraries(npgbench PRIVATE ggml ggml-base ggml-cpu ggml-hexagon log dl)
  target_compile_options(npgbench PRIVATE -fvisibility=hidden)
  target_link_options(npgbench PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/npg.map" "-Wl,-s" "-Wl,-z,max-page-size=16384")
  # HTP skels: prebuilt from the same unchanged ggml-hexagon sources (see scripts/build-android.sh)
else()
  add_executable(npgbench_host ${NPG_SRC})
  target_include_directories(npgbench_host PRIVATE ${CL_INC} "${SDCPP_DIR}/ggml/include")
  target_link_libraries(npgbench_host PRIVATE ggml ggml-base ggml-cpu dl pthread)
endif()
